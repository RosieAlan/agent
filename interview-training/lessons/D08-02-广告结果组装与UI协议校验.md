## app/schemas/ui.py（新建）

```python
from typing import Annotated, Literal
from pydantic import BaseModel, ConfigDict, Field


class UIModel(BaseModel):
    model_config = ConfigDict(strict=True, extra="forbid")  # 统一严格检查字段。


class CardField(UIModel):
    label: str = Field(min_length=1)  # 页面显示的字段名。
    value: str  # 已格式化的展示文字。


class CardComponent(UIModel):
    type: Literal["card"] = "card"  # 这个组件只允许card类型。
    title: str = Field(min_length=1)
    fields: list[CardField] = Field(min_length=1, max_length=20)


class SelectionOption(UIModel):
    id: str = Field(min_length=1)  # 后端识别选项的固定编号。
    label: str = Field(min_length=1)  # 用户看到的文案。


class SelectionComponent(UIModel):
    type: Literal["selection"] = "selection"
    title: str = Field(min_length=1)
    options: list[SelectionOption] = Field(min_length=1, max_length=10)
    allowMultiple: bool = False  # 与已有前端协议保持一致。


UIComponent = Annotated[
    CardComponent | SelectionComponent,  # 当前支持这两种组件。
    Field(discriminator="type"),  # 先读type，再按对应模型检查。
]


class UIResponse(UIModel):
    version: Literal["1.0"] = "1.0"  # 标明本版协议。
    message: str = Field(min_length=1)
    components: list[UIComponent] = Field(min_length=1, max_length=10)
```

## app/services/ui_service.py（新建）

```python
from app.repositories.ad_repository import get_ad_report_from_db
from app.schemas.ui import (
    UIResponse, CardComponent, CardField,
    SelectionComponent, SelectionOption,
)


class AdReportNotFound(Exception):
    pass  # HTTP入口将它转为404。


def build_ad_report_ui(user_id: str, store_id: str, campaign_id: str) -> UIResponse:
    report = get_ad_report_from_db(user_id, store_id, campaign_id)  # 沿用归属过滤。
    if report is None:
        raise AdReportNotFound()

    spend = report["spend"]
    sales = report["attributed_sales"]
    roas = f"{sales / spend:.2f}" if spend > 0 else "暂无：花费为0"  # 避免除零。
    currency = report["currency"]

    card = CardComponent(
        title=f"广告活动 {report['campaign_id']}",
        fields=[
            CardField(label="报表日期", value=report["report_date"]),
            CardField(label="花费", value=f"{spend:.2f} {currency}"),
            CardField(label="归因销售额", value=f"{sales:.2f} {currency}"),
            CardField(label="ROAS", value=roas),
        ],
    )  # 金额和日期来自数据库，创建对象时立即校验结构。
    selection = SelectionComponent(
        title="接下来做什么？",
        options=[
            SelectionOption(id="generate_report", label="生成分析报告"),
            SelectionOption(id="explain_metrics", label="查看指标说明"),
        ],
    )  # 选项由后端白名单定义；本块先输出，动作处理在后续接入。
    return UIResponse(
        message=f"已查询广告报表。数据来源：{report['data_source']}。",
        components=[card, selection],
    )  # 整份响应继续检查组件类型与数量。
```

## app/api/ui.py（新建）

```python
import logging
from fastapi import APIRouter, HTTPException
from pydantic import ValidationError
from sqlalchemy.exc import OperationalError, SQLAlchemyError, TimeoutError as PoolTimeoutError
from app.schemas.ui import UIResponse
from app.services.ui_service import build_ad_report_ui, AdReportNotFound

router = APIRouter(prefix="/ui", tags=["ui"])
logger = logging.getLogger(__name__)


@router.get("/ads/report/{campaign_id}", response_model=UIResponse)
def ad_report_ui(campaign_id: str):
    try:
        return build_ad_report_ui(
            user_id="user-001",  # 当前本地演练身份，正式登录接入后替换来源。
            store_id="store-A",  # 与当前工具演练店铺一致。
            campaign_id=campaign_id,  # URL里传来的活动编号。
        )
    except AdReportNotFound as error:
        raise HTTPException(404, "未找到可访问的广告报表") from error
    except (OperationalError, PoolTimeoutError) as error:
        raise HTTPException(503, "报表数据库暂时不可用") from error
    except SQLAlchemyError as error:
        logger.exception("UI report database query failed")
        raise HTTPException(500, "报表查询失败") from error
    except ValidationError as error:
        logger.exception("UI report structure invalid")
        raise HTTPException(500, "展示数据格式异常") from error
```

## 必背与题目答案

Q：discriminator='type'做什么？A：先读取组件type，card按CardComponent校验，selection按SelectionComponent校验；未知类型或缺必需字段会被拒绝。

Q：response_model和服务构造模型分别作用？A：服务构造时尽早检查组装结果；response_model规定最终HTTP响应的结构、文档和序列化，服务返回错误结构时也能拦住。

Q：为什么这里一次查询就能组装，不调用模型？A：报表字段与固定选项已经确定，后端直接生成协议；后续开放式分析交给模型，再校验返回结构。

Q：类型正确的9999元能通过Schema吗？A：展示字符串可通过结构校验，事实核对仍需要可信查询结果。本块金额直接由查询结果格式化，避免模型补写金额。

当前仍固定演示身份；数据源为合成，金额float用于展示，真实结算使用Decimal/整数分；完整JWT/工具授权、UI交互状态、动作幂等和其他组件类型按后续D08及安全单元展开。数据库结构/数据无需改变，无供应商调用。
